cmake_minimum_required(VERSION 3.16)
project(plan_predictive_cost_map LANGUAGES NONE)

find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)

ament_python_install_package(${PROJECT_NAME})
set(PYTHON_EXECUTABLES pcm_longitudinal pcm_costmap_viewer)
foreach(NODE_NAME IN LISTS PYTHON_EXECUTABLES)
  configure_file(tools/python_node.in ${CMAKE_CURRENT_BINARY_DIR}/${NODE_NAME} @ONLY)
  install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/${NODE_NAME} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(FILES launch/pcm_longitudinal.launch.py DESTINATION share/${PROJECT_NAME}/launch)
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(longitudinal test/test_pcm_longitudinal_ros.py TIMEOUT 900)
  ament_add_pytest_test(light_corners test/test_pcm_light_corners.py TIMEOUT 120)
  ament_add_pytest_test(sqp test/test_pcm_sqp.py TIMEOUT 60)
endif()
ament_package()
