cmake_minimum_required(VERSION 3.16)
project(plan_predictive_cost_map)

find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(crp_msgs REQUIRED)
find_package(autoware_control_msgs REQUIRED)
find_package(autoware_planning_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(NLopt CONFIG REQUIRED PATHS "$ENV{HOME}/.cache/pcm_native/install/lib/cmake/nlopt")
if(NOT EXISTS "$ENV{HOME}/.cache/pcm_native_json.hpp")
  message(FATAL_ERROR "Pinned native JSON header is missing from ~/.cache/pcm_native_json.hpp")
endif()
configure_file("$ENV{HOME}/.cache/pcm_native_json.hpp" "${CMAKE_CURRENT_BINARY_DIR}/native_json.hpp" COPYONLY)
execute_process(COMMAND ${Python3_EXECUTABLE} -c "import casadi,pathlib; print(pathlib.Path(casadi.__file__).parent / 'cmake')"
  OUTPUT_VARIABLE CASADI_CMAKE_PATH OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
find_package(casadi CONFIG REQUIRED PATHS "${CASADI_CMAKE_PATH}" NO_DEFAULT_PATH)

ament_python_install_package(${PROJECT_NAME})
set(PYTHON_EXECUTABLES pcm mppc pcm_combined pcm_longitudinal pcm_costmap_viewer pcm_planner mpc_tracker safety_supervisor)
foreach(NODE_NAME IN LISTS PYTHON_EXECUTABLES)
  configure_file(tools/python_node.in ${CMAKE_CURRENT_BINARY_DIR}/${NODE_NAME} @ONLY)
  install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/${NODE_NAME} DESTINATION lib/${PROJECT_NAME})
endforeach()
add_executable(pcm_longitudinal_cpp src/pcm_longitudinal_cpp.cpp)
add_library(pcm_casadi_bridge SHARED src/casadi_bridge.cpp)
target_link_libraries(pcm_casadi_bridge PRIVATE casadi::casadi)
set_target_properties(pcm_casadi_bridge PROPERTIES INSTALL_RPATH "${CASADI_CMAKE_PATH}/..")
target_compile_features(pcm_longitudinal_cpp PRIVATE cxx_std_17)
target_include_directories(pcm_longitudinal_cpp PRIVATE include ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(pcm_longitudinal_cpp Eigen3::Eigen pcm_casadi_bridge NLopt::nlopt)
set_target_properties(pcm_longitudinal_cpp PROPERTIES INSTALL_RPATH "$ORIGIN/..;$ENV{HOME}/.cache/pcm_native/install/lib")
ament_target_dependencies(pcm_longitudinal_cpp rclcpp crp_msgs autoware_control_msgs autoware_planning_msgs std_msgs)
install(TARGETS pcm_longitudinal_cpp DESTINATION lib/${PROJECT_NAME})
install(TARGETS pcm_casadi_bridge DESTINATION lib)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES resource/acados_stage_runtime.c.in DESTINATION share/${PROJECT_NAME}/resource)
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(native_gp test/test_pcm_native_gp.py TIMEOUT 60)
  ament_add_pytest_test(native_reference test/test_pcm_native_reference.py TIMEOUT 60)
  ament_add_pytest_test(native_live test/test_pcm_native_live.py TIMEOUT 180)
  ament_add_pytest_test(sqp test/test_pcm_sqp.py TIMEOUT 60)
  ament_add_pytest_test(native_solver test/test_pcm_native_solver.py TIMEOUT 60)
  ament_add_pytest_test(acados_stages test/test_pcm_acados_stages.py TIMEOUT 180)
endif()
ament_package()
